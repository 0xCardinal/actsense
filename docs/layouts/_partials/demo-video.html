{{- /* The demo loop recorded by docs/scripts/record_demo.py, as a light and a dark video that follow the docs theme. static/demo-video.js decides which one loads and plays. Pass a dict with `class` for the video elements. */ -}} {{- $class := .class | default "" -}} {{- range slice (dict "theme" "light" "suffix" "") (dict "theme" "dark" "suffix" "-dark") }} {{- $base := printf "videos/demo%s" .suffix -}} {{- $webm := resources.Get (printf "%s.webm" $base) | fingerprint "md5" -}} {{- $mp4 := resources.Get (printf "%s.mp4" $base) | fingerprint "md5" -}} {{- $poster := resources.Get (printf "%s.webp" $base) | fingerprint "md5" }} {{- end }}
